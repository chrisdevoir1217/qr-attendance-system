let lastScan = ""; function onScanSuccess(decodedText) { if (decodedText === lastScan) return; lastScan = decodedText; const status = document.getElementById("status"); // 🔵 scanning detected status.innerText = "Scanning..."; status.style.background = "#ffe08a"; // yellow fetch(webhookURL, { method: "POST", headers: {"Content-Type": "application/json"}, body: JSON.stringify({ student_id: decodedText, timestamp: new Date().toISOString() }) }) .then(res => { if (res.ok) { // 🟢 SUCCESS status.innerText = "✅ DONE - QR SENT"; status.style.background = "#7CFC90"; // green // reset after 2 seconds setTimeout(() => { status.innerText = "Waiting for scan..."; status.style.background = "#eee"; }, 2000); } else { // 🔴 server error status.innerText = "❌ ERROR - NOT SENT"; status.style.background = "#ff6b6b"; } }) .catch(err => { status.innerText = "❌ NETWORK ERROR"; status.style.background = "#ff6b6b"; console.log(err); }); }
